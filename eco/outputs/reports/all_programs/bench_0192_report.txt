        avg = (w0 * x0 + w1 * x1 + w2 * x2 + w3 * x3) // (w0 + w1 + w2 + w3)
        total_weight = w0 + w1 + w2 + w3
        scaled = (w0 * x0 + w1 * x1 + w2 * x2 + w3 * x3) // total_weight

    def cross_product_3d(ax, ay, az, bx, by, bz):
        cx = ay * bz - az * by
        cy = az * bx - ax * bz
        cz = ax * by - ay * bx
        norm_sq = (ay * bz - az * by) * (ay * bz - az * by) + cy * cy + cz * cz
        return cx, cy, cz, norm_sq
